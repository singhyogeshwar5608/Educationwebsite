    34→// ─── Generate Certificate Number ────────────────────────────
    35→function generateCertificateNo(existingCerts: Certificate[]): string {
    36→  const year = new Date().getFullYear()
    37→  const existingNums = existingCerts
    38→    .filter((c) => c.certificateNo.startsWith(`ZTECH-${year}-`))
    39→    .map((c) => {
    40→      const parts = c.certificateNo.split('-')
    41→      return parseInt(parts[2], 10) || 0
    42→    })
    43→  const nextSeq = existingNums.length > 0 ? Math.max(...existingNums) + 1 : 1
    44→  return `ZTECH-${year}-${String(nextSeq).padStart(3, '0')}`
    45→}

    47→// ─── Step indicator ─────────────────────────────────────────
    48→function CertStepIndicator({ currentStep }: { currentStep: number }) {

    56→      {steps.map((step, idx) => {
    57→        const stepNum = idx + 1
    58→        const isActive = stepNum === currentStep

    60→        const Icon = step.icon
    61→        return (
    62→          <div key={stepNum} className="flex items-center">
    63→            <div className="flex flex-col items-center">
    64→              <div
    65→                className={`w-12 h-12 rounded-full flex items-center justify-center text-sm font-bold transition-all duration-300 ${
    66→                  isCompleted

    69→                    ? 'bg-gold text-navy shadow-lg shadow-gold/30'
    70→                    : 'bg-gray-200 text-text-gray'
    71→                }`}

    97→// ─── Grade color helper ─────────────────────────────────────
    98→function getGradeColor(grade: string): string {
    99→  if (grade === 'A+' || grade === 'A') return 'text-green font-bold'
   100→  if (grade === 'B+' || grade === 'B') return 'text-blue-600 font-bold'
   101→  if (grade === 'C' || grade === 'D') return 'text-amber-600 font-bold'
   102→  if (grade === 'F') return 'text-red-600 font-bold'
   103→  return 'text-navy font-bold'
   104→}
   105→
   106→function getGradeBadgeClass(grade: string): string {
   107→  if (grade === 'A+' || grade === 'A') return 'badge-success'
   108→  if (grade === 'B+' || grade === 'B') return 'badge-info'
   109→  if (grade === 'C' || grade === 'D') return 'badge-warning'
   110→  if (grade === 'F') return 'badge-danger'
   111→  return 'badge-info'
   112→}
